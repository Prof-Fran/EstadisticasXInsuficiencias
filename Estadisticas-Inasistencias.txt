    /* VARIABLES DE CONTROL E INSTANCIAS */
    let currentGrade = "7";
    let activeGroupIndex = 0;
    let activeMetricSelected = "sinBajas"; // Categoría seleccionada por defecto

    let activeDonutChart = null;
    let activeBarChart = null;

    // Diagnósticos guardados temporalmente en memoria para evitar reseteos al cambiar de pantalla
    const savedInsights = {};

    /* ELEMENTOS DEL DOM */
    const tabButtons = document.querySelectorAll('.tab-btn');
    const groupSelect = document.getElementById('group-select');
    const totalGruposEl = document.getElementById('total-grupos');
    const totalSinBajasEl = document.getElementById('total-sin-bajas');
    const totalCuatroMasEl = document.getElementById('total-cuatro-mas');
    const totalAusentismoCriticoEl = document.getElementById('total-ausentismo-critico');

    const focusGroupNameEl = document.getElementById('focus-group-name');
    const focusGroupTotalStudentsEl = document.getElementById('focus-group-total-students');
    const donutCenterPctEl = document.getElementById('donut-center-pct');

    const valSinBajasEl = document.getElementById('val-sinBajas');
    const valEntre1y3El = document.getElementById('val-entre1y3');
    const valCuatroOMasEl = document.getElementById('val-cuatroOMas');

    const comparativeGridContainer = document.getElementById('comparative-grid-container');
    const insightsTextEl = document.getElementById('insights-text');
    const printReportTitle = document.getElementById('print-report-title');

    // Elementos del bloque de ausentismo dinámico
    const absencesBoxTitle = document.getElementById('absences-box-title');
    const absenceValRegular = document.getElementById('absence-val-regular');
    const absenceValModerado = document.getElementById('absence-val-moderado');
    const absenceValCritico = document.getElementById('absence-val-critico');

    const metricItems = document.querySelectorAll('.focus-metric-item');

    /* LÓGICA DE INICIALIZACIÓN */
    window.addEventListener('DOMContentLoaded', () => {
      setupGrade(currentGrade);
      setupMetricInteractivity();
    });

    // Cambiar de grado
    tabButtons.forEach(btn => {
      btn.addEventListener('click', (e) => {
        tabButtons.forEach(b => {
          b.classList.remove('active');
          b.setAttribute('aria-selected', 'false');
        });
        btn.classList.add('active');
        btn.setAttribute('aria-selected', 'true');
        
        currentGrade = btn.getAttribute('data-tab');
        activeGroupIndex = 0;
        activeMetricSelected = "sinBajas"; // resetear a por defecto al cambiar grado
        setupGrade(currentGrade);
      });
    });

    // Selector Select manual
    groupSelect.addEventListener('change', (e) => {
      saveCurrentGroupInsights();
      activeGroupIndex = parseInt(e.target.value, 10);
      renderFocusCard();
    });

    /* CONFIGURACIÓN DEL GRADO */
    function setupGrade(grade) {
      saveCurrentGroupInsights();
      const groups = dashboardData[grade] || [];
      
      // Actualizar Bento Global Estadístico del Grado Completo
      totalGruposEl.textContent = groups.length;
      
      let grandTotalStudents = 0;
      let totalSinBajasGrade = 0;
      let totalCuatroMasGrade = 0;
      let totalCriticoGrade = 0; // Para calcular la cuarta tarjeta

      groups.forEach(g => {
        const totalG = g.sinBajas + g.entre1y3 + g.cuatroOMas;
        grandTotalStudents += totalG;
        totalSinBajasGrade += g.sinBajas;
        totalCuatroMasGrade += g.cuatroOMas;

        // Sumar críticos de todas las categorías del grupo
        if (g.distribucionInasistencias) {
          totalCriticoGrade += g.distribucionInasistencias.sinBajas.critico +
                               g.distribucionInasistencias.entre1y3.critico +
                               g.distribucionInasistencias.cuatroOMas.critico;
        }
      });

      if (grandTotalStudents > 0) {
        totalSinBajasEl.textContent = ((totalSinBajasGrade / grandTotalStudents) * 100).toFixed(1) + "%";
        totalCuatroMasEl.textContent = ((totalCuatroMasGrade / grandTotalStudents) * 100).toFixed(1) + "%";
        totalAusentismoCriticoEl.textContent = ((totalCriticoGrade / grandTotalStudents) * 100).toFixed(1) + "%";
      } else {
        totalSinBajasEl.textContent = "0.0%";
        totalCuatroMasEl.textContent = "0.0%";
        totalAusentismoCriticoEl.textContent = "0.0%";
      }

      // Reestablecer Select de opciones
      groupSelect.innerHTML = "";
      groups.forEach((g, idx) => {
        const opt = document.createElement('option');
        opt.value = idx;
        opt.textContent = `Grupo ${g.name}`;
        if (idx === activeGroupIndex) opt.selected = true;
        groupSelect.appendChild(opt);
      });

      // Renderizar grid rápida y enfoque principal
      renderQuickGrid(groups);
      renderFocusCard();
    }

    /* RENDER DE LA LISTA RÁPIDA INFERIOR */
    function renderQuickGrid(groups) {
      comparativeGridContainer.innerHTML = "";
      groups.forEach((g, idx) => {
        const totalG = g.sinBajas + g.entre1y3 + g.cuatroOMas;
        const pSinBajas = totalG > 0 ? (g.sinBajas / totalG) * 100 : 0;
        const pMedio = totalG > 0 ? (g.entre1y3 / totalG) * 100 : 0;
        const pCritico = totalG > 0 ? (g.cuatroOMas / totalG) * 100 : 0;

        const card = document.createElement('article');
        card.className = `comparative-card ${idx === activeGroupIndex ? 'active' : ''}`;
        card.setAttribute('tabindex', '0');
        card.innerHTML = `
          <div class="comparative-card-header">
            <span class="comparative-card-title">Grupo ${g.name}</span>
            <span class="comparative-card-badge">${totalG} Alumn.</span>
          </div>
          <div class="comparative-mini-bar">
            <div class="mini-bar-segment mini-bar-success" style="width: ${pSinBajas}%"></div>
            <div class="mini-bar-segment mini-bar-warning" style="width: ${pMedio}%"></div>
            <div class="mini-bar-segment mini-bar-danger" style="width: ${pCritico}%"></div>
          </div>
        `;

        card.addEventListener('click', () => {
          saveCurrentGroupInsights();
          activeGroupIndex = idx;
          groupSelect.value = idx;
          
          document.querySelectorAll('.comparative-card').forEach(c => c.classList.remove('active'));
          card.classList.add('active');
          
          renderFocusCard();
        });

        comparativeGridContainer.appendChild(card);
      });
    }

    /* INTERACTIVIDAD CRUZADA DE LA LISTA DE METRICAS (BOTONES DE FILTRO) */
    function setupMetricInteractivity() {
      metricItems.forEach(item => {
        item.addEventListener('click', () => {
          metricItems.forEach(i => {
            i.classList.remove('selected-active');
            i.setAttribute('aria-pressed', 'false');
          });
          item.classList.add('selected-active');
          item.setAttribute('aria-pressed', 'true');

          activeMetricSelected = item.getAttribute('data-metric');
          updateAbsencesBreakdownBlock();
        });
      });
    }

    /* RENDERIZADO DEL ENFOQUE PRINCIPAL Y SUS COMPONENTES GRÁFICOS */
    function renderFocusCard() {
      const gradeStr = currentGrade === "EMS" ? "1° EMS" : `${currentGrade}° Año`;
      printReportTitle.textContent = `Reporte de Rendimiento Académico y Ausentismo - ${gradeStr}`;

      const groups = dashboardData[currentGrade] || [];
      const group = groups[activeGroupIndex];
      if (!group) return;

      // Actualizar etiquetas básicas
      focusGroupNameEl.textContent = `Grupo: ${group.name}`;
      const totalStudents = group.sinBajas + group.entre1y3 + group.cuatroOMas;
      focusGroupTotalStudentsEl.textContent = `${totalStudents} Alumnos`;

      valSinBajasEl.textContent = group.sinBajas;
      valEntre1y3El.textContent = group.entre1y3;
      valCuatroOMasEl.textContent = group.cuatroOMas;

      // Calcular porcentaje del donut (Alumnos con 0-3 materias pendientes)
      const promotedCount = group.sinBajas + group.entre1y3;
      const pctPromoted = totalStudents > 0 ? ((promotedCount / totalStudents) * 100).toFixed(0) : 0;
      donutCenterPctEl.textContent = `${pctPromoted}%`;

      // Cargar diagnóstico guardado o dejar vacío
      const insightKey = `${currentGrade}_${group.name}`;
      insightsTextEl.innerHTML = savedInsights[insightKey] || "";

      // Actualizar Bloques Gráficos y de Ausentismo Dinámico
      updateDonutChartInstance(group);
      updateBarChartInstance(group);
      updateAbsencesBreakdownBlock();
    }

    /* ACTUALIZACIÓN DEL CONTENEDOR DE AUSENTISMO DINÁMICO SEGÚN LA CATEGORÍA CLICADA */
    function updateAbsencesBreakdownBlock() {
      const groups = dashboardData[currentGrade] || [];
      const group = groups[activeGroupIndex];
      if (!group || !group.distribucionInasistencias) return;

      let categoryLabel = "Sin asignaturas pendientes";
      if (activeMetricSelected === "entre1y3") categoryLabel = "Rendimiento Medio (1-3 pend.)";
      if (activeMetricSelected === "cuatroOMas") categoryLabel = "Rendimiento Crítico (4+ pend.)";

      absencesBoxTitle.textContent = `Niveles de Ausentismo en: ${categoryLabel}`;

      const dataFaltas = group.distribucionInasistencias[activeMetricSelected];
      if (dataFaltas) {
        absenceValRegular.textContent = `${dataFaltas.regular} Alumn.`;
        absenceValModerado.textContent = `${dataFaltas.moderado} Alumn.`;
        absenceValCritico.textContent = `${dataFaltas.critico} Alumn.`;
      }
    }

    /* SALVAGUARDAR DIAGNÓSTICO EN MEMORIA */
    function saveCurrentGroupInsights() {
      const groups = dashboardData[currentGrade] || [];
      const group = groups[activeGroupIndex];
      if (group) {
        const insightKey = `${currentGrade}_${group.name}`;
        savedInsights[insightKey] = insightsTextEl.innerHTML;
      }
    }

    /* INSTANCIACIÓN Y ACTUALIZACIÓN DEL GRÁFICO DONUT (CHART.JS) */
    function updateDonutChartInstance(group) {
      const ctx = document.getElementById('donutChart').getContext('2d');
      
      const chartData = [group.sinBajas, group.entre1y3, group.cuatroOMas];

      if (activeDonutChart) {
        activeDonutChart.data.datasets[0].data = chartData;
        activeDonutChart.update();
      } else {
        activeDonutChart = new Chart(ctx, {
          type: 'doughnut',
          data: {
            labels: ['Sin Asig. Pendientes', 'Rendimiento Medio', 'Rendimiento Crítico'],
            datasets: [{
              data: chartData,
              backgroundColor: ['#00f2fe', '#f59e0b', '#ff007f'],
              borderWidth: 2,
              borderColor: '#07070a',
              hoverOffset: 4
            }]
          },
          options: {
            cutout: '76%',
            responsive: true,
            maintainAspectRatio: false,
            plugins: {
              legend: { display: false },
              datalabels: { display: false }
            }
          }
        });
      }
    }

    /* INSTANCIACIÓN Y ACTUALIZACIÓN DEL HISTOGRAMA HORIZONTAL (CHART.JS) */
    function updateBarChartInstance(group) {
      const ctx = document.getElementById('barChart').getContext('2d');
      
      const subjects = Object.keys(group.insuficiencias);
      const values = Object.values(group.insuficiencias);

      if (activeBarChart) {
        activeBarChart.data.labels = subjects;
        activeBarChart.data.datasets[0].data = values;
        activeBarChart.update();
      } else {
        activeBarChart = new Chart(ctx, {
          type: 'bar',
          data: {
            labels: subjects,
            datasets: [{
              label: 'Alumnos con insuficiencia',
              data: values,
              backgroundColor: 'rgba(99, 102, 241, 0.35)',
              borderColor: '#818cf8',
              borderWidth: 2,
              borderRadius: 6,
              borderSkipped: false,
              barPercentage: 0.65
            }]
          },
          options: {
            indexAxis: 'y',
            responsive: true,
            maintainAspectRatio: false,
            scales: {
              x: {
                grid: { color: 'rgba(255, 255, 255, 0.05)' },
                ticks: { color: '#9ca3af', font: { family: 'Inter', size: 11 }, stepSize: 1 }
              },
              y: {
                grid: { display: false },
                ticks: { color: '#ffffff', font: { family: 'Outfit', weight: '600', size: 12 } }
              }
            },
            plugins: {
              legend: { display: false },
              datalabels: {
                display: true,
                anchor: 'end',
                align: 'right',
                color: '#00f2fe',
                font: { family: 'Outfit', weight: 'bold', size: 12 },
                formatter: (val) => val > 0 ? val : ''
              }
            }
          }
        });
      }
    }

      saveCurrentGroupInsights();
      window.print();

    window.addEventListener('beforeprint', () => {
      if (activeDonutChart) {
        activeDonutChart.data.datasets[0].borderColor = '#ffffff';
        activeDonutChart.update('none');
      }
      if (activeBarChart) {
        activeBarChart.options.scales.y.ticks.color = '#111827';
        activeBarChart.options.scales.x.ticks.color = '#4b5563';
        activeBarChart.options.scales.x.grid.color = 'rgba(0, 0, 0, 0.1)';
        if (activeBarChart.options.plugins.datalabels) {
          activeBarChart.options.plugins.datalabels.color = '#0369a1';
        }
        activeBarChart.update('none');
      }
    });

    window.addEventListener('afterprint', () => {
      if (activeDonutChart) {
        activeDonutChart.data.datasets[0].borderColor = '#07070a';
        activeDonutChart.update('none');
      }
      if (activeBarChart) {
        activeBarChart.options.scales.y.ticks.color = '#ffffff';
        activeBarChart.options.scales.x.ticks.color = '#9ca3af';
        activeBarChart.options.scales.x.grid.color = 'rgba(255, 255, 255, 0.05)';
        if (activeBarChart.options.plugins.datalabels) {
          activeBarChart.options.plugins.datalabels.color = '#00f2fe';
        }
        activeBarChart.update('none');
      }
    });
